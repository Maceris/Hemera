# HMIR lib

add_mlir_dialect_library(HMIR
  ${HEMERA_MLIR_DIR}/lib/hmir/hmir_dialect.cpp
  ${HEMERA_MLIR_DIR}/lib/hmir/hmir_passes.cpp
  ${HEMERA_MLIR_DIR}/lib/hmir/ir/hmir_ops.cpp

  DEPENDS
  MLIRhmir_opsIncGen
  MLIRhmir_passesIncGen

  LINK_COMPONENTS
  Core

  LINK_LIBS PUBLIC
  MLIRIR
  MLIRInferTypeOpInterface
  MLIRFuncDialect
  MLIRLLVMDialect
)

TARGET_INCLUDE_DIRECTORIES(HMIR PUBLIC
  $<BUILD_INTERFACE:${INCLUDE_DIR}/>
  $<BUILD_INTERFACE:${HEMERA_MLIR_DIR}/include>
)

TARGET_LINK_LIBRARIES(HMIR PUBLIC MLIRIR)
TARGET_COMPILE_OPTIONS(HMIR PRIVATE ${HEMERA_OPTIONS})
TARGET_COMPILE_DEFINITIONS(HMIR PRIVATE ${HEMERA_DEFINITIONS})

# A folder per configuration: LLVM's output directory, which add_mlir_dialect_library uses, is set to one without the
# configuration in it, so otherwise a Debug and a Release build overwrite each other's library and fail to link
FOREACH(CONFIGURATION ${CMAKE_CONFIGURATION_TYPES})
  STRING(TOUPPER ${CONFIGURATION} CONFIGURATION_UPPER)
  SET_TARGET_PROPERTIES(HMIR PROPERTIES
    ARCHIVE_OUTPUT_DIRECTORY_${CONFIGURATION_UPPER} ${HEMERA_LIBRARY_OUTPUT_DIRECTORY}/${CONFIGURATION}
    LIBRARY_OUTPUT_DIRECTORY_${CONFIGURATION_UPPER} ${HEMERA_LIBRARY_OUTPUT_DIRECTORY}/${CONFIGURATION}
    RUNTIME_OUTPUT_DIRECTORY_${CONFIGURATION_UPPER} ${HEMERA_RUNTIME_OUTPUT_DIRECTORY}/${CONFIGURATION})
ENDFOREACH()
